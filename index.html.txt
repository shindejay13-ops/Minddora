  <script>
    /* ========== MOBILE MENU ========== */
    const menuBtn = document.getElementById('menuBtn');
    const navLinks = document.getElementById('navLinks');
    menuBtn.addEventListener('click', () => {
      const isOpen = navLinks.classList.toggle('open');
      menuBtn.setAttribute('aria-expanded', isOpen);
    });
    navLinks.querySelectorAll('a').forEach(link => {
      link.addEventListener('click', () => {
        navLinks.classList.remove('open');
        menuBtn.setAttribute('aria-expanded', 'false');
      });
    });
    /* ========== MOOD TRACKER ========== */
    const moodResponses = {
      happy: "That's lovely to hear  Keep noticing what is going well — savour the good moments.",
      calm: "A calm mind is a strong mind  Try to stay with this feeling for a few more breaths.",
      sad: "It's okay to feel sad sometimes  Be gentle with yourself. A short walk, water, or talking to someone you trust can help.",
      anxious: "Anxiety can feel heavy ️ Try the breathing exercise on this page — slow breaths signal safety to your body.",
      angry: "Anger is a valid emotion  Pause, breathe out slowly, and give yourself a moment before reacting.",
      tired: "Rest is productive  Your body is asking for a break. Even a 5-minute pause or a glass of water helps."
    };
    const moodGrid = document.getElementById('moodGrid');
    const moodResponse = document.getElementById('moodResponse');
    moodGrid.addEventListener('click', (e) => {
      const btn = e.target.closest('.mood-btn');
      if (!btn) return;
      const mood = btn.dataset.mood;
      moodGrid.querySelectorAll('.mood-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      const msg = moodResponses[mood] || "Thank you for checking in ";
      moodResponse.innerHTML = `<p>${msg}</p>`;
    });
    /* ========== JOURNAL ========== */
    const journalText = document.getElementById('journalText');
    const saveJournal = document.getElementById('saveJournal');
    const clearJournal = document.getElementById('clearJournal');
    const journalStatus = document.getElementById('journalStatus');
    const STORAGE_KEY = 'mindora_journal_note';
    // Load saved note on page load
    if (localStorage.getItem(STORAGE_KEY)) {
      journalText.value = localStorage.getItem(STORAGE_KEY);
    }
    saveJournal.addEventListener('click', () => {
      const text = journalText.value.trim();
      if (!text) {
        journalStatus.textContent = 'Please write something before saving.';
        journalStatus.style.color = '#c0392b';
        return;
      }
      localStorage.setItem(STORAGE_KEY, text);
      journalStatus.textContent = 'Saved on this device only. ✅';
      journalStatus.style.color = '#2e7d32';
    });
    clearJournal.addEventListener('click', () => {
      journalText.value = '';
      localStorage.removeItem(STORAGE_KEY);
      journalStatus.textContent = 'Cleared.';
      journalStatus.style.color = '#6a5acd';
    });
    /* ========== BREATHING EXERCISE ========== */
    const breathBtn = document.getElementById('breathBtn');
    const breathCircle = document.getElementById('breathCircle');
    const breathLabel = document.getElementById('breathLabel');
    const breathCount = document.getElementById('breathCount');
    let breathing = false;
    let breathTimer = null;
    let phaseTimer = null;
    const phases = [
      { name: 'Breathe in', className: 'inhale', seconds: 4 },
      { name: 'Hold', className: 'hold', seconds: 4 },
      { name: 'Breathe out', className: 'exhale', seconds: 4 }
    ];
    function stopBreathing() {
      breathing = false;
      breathBtn.textContent = 'Start';
      clearTimeout(breathTimer);
      clearInterval(phaseTimer);
      breathCircle.className = 'breath-ring';
      breathLabel.textContent = 'Ready';
      breathCount.textContent = '—';
    }
    function runPhase(index) {
      if (!breathing) return;
      const phase = phases[index];
      breathCircle.className = 'breath-ring ' + phase.className;
      breathLabel.textContent = phase.name;
      let remaining = phase.seconds;
      breathCount.textContent = remaining;
      clearInterval(phaseTimer);
      phaseTimer = setInterval(() => {
        remaining--;
        if (remaining > 0) {
          breathCount.textContent = remaining;
        } else {
          clearInterval(phaseTimer);
          breathCount.textContent = '0';
          const next = (index + 1) % phases.length;
          breathTimer = setTimeout(() => runPhase(next), 200);
        }
      }, 1000);
    }
    breathBtn.addEventListener('click', () => {
      if (breathing) {
        stopBreathing();
      } else {
        breathing = true;
        breathBtn.textContent = 'Stop';
        runPhase(0);
      }
    });
  </script>
